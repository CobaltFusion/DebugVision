# debugvision.exe as one self-contained executable: the application, the CopperSpice libraries, the platform plugin
# and the MSVC runtime are embedded as compressed resources, unpacked into a temporary directory at startup and removed again
# when the application exits (see main.cpp).

get_target_property(DV_CSCORE_DLL CopperSpice::CsCore LOCATION)
get_target_property(DV_CSGUI_DLL CopperSpice::CsGui LOCATION)

# pairs of <path in the unpacked directory> <source file>
set(DV_PAYLOAD
  debugvision.exe $<TARGET_FILE:debugvision>
  CsCore${COPPERSPICE_VERSION_API}.dll ${DV_CSCORE_DLL}
  CsGui${COPPERSPICE_VERSION_API}.dll ${DV_CSGUI_DLL}
  platforms/CsGuiWin${COPPERSPICE_VERSION_API}.dll ${CS_PLUGIN_DIR}/CsGuiWin${COPPERSPICE_VERSION_API}.dll
)

# the MSVC runtime, so no Visual C++ redistributable needs to be installed (the UCRT is part of Windows 10 and later)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
include(InstallRequiredSystemLibraries)
foreach(lib ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS})
  get_filename_component(name ${lib} NAME)
  string(TOLOWER ${name} name)
  if (name MATCHES "^(msvcp140|vcruntime140|vcruntime140_1)\\.dll$")
    list(APPEND DV_PAYLOAD ${name} ${lib})
  endif()
endforeach()

# build tool that compresses the embedded files
add_executable(dv_compress compress.cpp)
target_link_libraries(dv_compress PRIVATE Cabinet project::warnings_options project::warning_suppressions project::project_options)

# collect the compressed payload in one directory, so the resource compiler reruns when one of the files changes
set(DV_PAYLOAD_DIR ${CMAKE_CURRENT_BINARY_DIR}/payload)
set(DV_PAYLOAD_RC "IDI_ICON1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/../application/icons/debugvision.ico\"\n")
set(DV_PAYLOAD_TABLE "")
set(DV_COMPRESS_COMMANDS "")
set(DV_PAYLOAD_SOURCES debugvision)
set(id 100)
list(LENGTH DV_PAYLOAD count)
math(EXPR last "${count} - 1")
foreach(i RANGE 0 ${last} 2)
  math(EXPR j "${i} + 1")
  list(GET DV_PAYLOAD ${i} path)
  list(GET DV_PAYLOAD ${j} source)
  string(APPEND DV_PAYLOAD_RC "${id} DVFILE \"${DV_PAYLOAD_DIR}/${path}\"\n")
  string(APPEND DV_PAYLOAD_TABLE "    {${id}, L\"${path}\"},\n")
  list(APPEND DV_COMPRESS_COMMANDS COMMAND dv_compress ${source} ${DV_PAYLOAD_DIR}/${path})
  list(APPEND DV_PAYLOAD_SOURCES ${source})
  math(EXPR id "${id} + 1")
endforeach()

file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/payload.rc CONTENT "${DV_PAYLOAD_RC}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/payload.h CONTENT
"// generated by src/launcher/CMakeLists.txt: the resource ids of the embedded files and their path in the unpacked directory
#pragma once

struct PayloadFile
{
    int id;
    const wchar_t* path;
};

inline constexpr PayloadFile payload[] = {
${DV_PAYLOAD_TABLE}};
")

# hash the embedded files on every build: payload.hash is rewritten only when their contents changed, so the files are
# compressed again even when a replaced file kept an older timestamp (e.g. after reinstalling CopperSpice)
set(DV_PAYLOAD_FILES ${DV_PAYLOAD_SOURCES})
list(REMOVE_ITEM DV_PAYLOAD_FILES debugvision) # the target name, its file is in the list as $<TARGET_FILE:debugvision>
list(JOIN DV_PAYLOAD_FILES "\n" DV_PAYLOAD_SOURCE_LINES)
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/payload_files_$<CONFIG>.txt CONTENT "${DV_PAYLOAD_SOURCE_LINES}\n")
add_custom_target(debugvision_payload_hash
  COMMAND ${CMAKE_COMMAND} -DFILE_LIST=${CMAKE_CURRENT_BINARY_DIR}/payload_files_$<CONFIG>.txt -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/payload.hash -P ${CMAKE_CURRENT_SOURCE_DIR}/payload_hash.cmake
  BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/payload.hash
  COMMENT "Checking the contents of the files embedded in debugvision.exe"
  VERBATIM
)
add_dependencies(debugvision_payload_hash debugvision)

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/payload.stamp
  COMMAND ${CMAKE_COMMAND} -E make_directory ${DV_PAYLOAD_DIR}/platforms
  ${DV_COMPRESS_COMMANDS}
  COMMAND ${CMAKE_COMMAND} -E touch ${CMAKE_CURRENT_BINARY_DIR}/payload.stamp
  DEPENDS ${DV_PAYLOAD_SOURCES} ${CMAKE_CURRENT_BINARY_DIR}/payload.hash dv_compress
  COMMENT "Compressing the files embedded in debugvision.exe"
)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/payload.rc PROPERTIES OBJECT_DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/payload.stamp)

add_executable(debugvision_launcher WIN32
  main.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/payload.rc
)

target_include_directories(debugvision_launcher PRIVATE ${CMAKE_CURRENT_BINARY_DIR})

# the launcher itself links the C++ runtime statically, it has to start before the embedded runtime is unpacked
set_target_properties(debugvision_launcher PROPERTIES
  OUTPUT_NAME debugvision
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
)

target_link_libraries(debugvision_launcher PRIVATE
  Cabinet
  project::warnings_options
  project::warning_suppressions
  project::project_options
)

install(TARGETS debugvision_launcher RUNTIME DESTINATION ${CMAKE_INSTALL_PREFIX})
